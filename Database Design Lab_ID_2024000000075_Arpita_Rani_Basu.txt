Create table Customer (   
Customer_ID Number(10),   
Customer_Name Varchar2(50),   
Email_Address Varchar2(100),   
Contect_N0 Varchar2(15),   
Customer_Address Varchar2(150),   
Primary Key (Customer_ID) );
  
DESC Customer;

Create table Category (
  Category_ID Number(10),
  Category_Name Varchar2(50),
  Primary Key (Category_ID)
);
DESC Category;

Create table Product (
  Product_ID Number(10),
  Product_Name varchar2(100),
  Price Number(10,2),
  Stock Number(10),
  category_ID Number(10),
  Primary Key (Product_ID),
  Foreign Key (Category_ID) References Category(Category_ID)
);
DESC product;

Create table Orders(
  Order_ID Number(10),
  Order_Date Date,
  Total_Amount Number (10,2),
  Customer_ID Number(10),
  Primary Key (Order_ID),
  Foreign Key (Customer_ID) References Customer(Customer_ID)
);
DESC Orders;

Create table Payment(
  Payment_ID Number(10),
  Payment_Method varchar2(30),
  Payment_Date Date,
  Amount Number(10,2),
  Order_ID Number(10),
  Primary Key (Payment_ID),
  Foreign Key (Order_ID) References Orders(Order_ID)
);
DESC Payment;

Create table Delivery (
Delivery_ID Number(10), 
Delivery_Name Varchar2(50),   
Salary Number(10,2),   
Phone Varchar2(15),   
Vehicle_No Varchar2(20),   
Order_ID Number(10),   
Primary Key (Delivery_ID),   
Foreign Key (Order_ID) References Orders(Order_ID) 
); 
DESC Delivery;

Create table Order_Item(
  Order_ID Number(10),
  Product_ID Number(10),
  Quantity Number (10),
  Subtotal Number(10,2),
  Primary Key (Order_ID, Product_ID),
  Foreign Key (Order_ID) References Orders(Order_ID),
  Foreign Key (Product_ID) References Product(Product_ID)
);
DESC Order_Item;

SELECT c.Customer_Name, o.Order_ID, o.Total_Amount
FROM Customer c

    SELECT MAX(Total_Amount)

--Find the most expensive product in each category
SELECT p.Product_ID,
       p.Product_Name,
       p.Category_ID,
       p.Price
FROM Product p
WHERE p.Price = (
    SELECT MAX(p2.Price)
    FROM Product p2
    WHERE p2.Category_ID = p.Category_ID
);

--Find payment methods whose total payment is greater than the average payment-method total
SELECT Payment_Method,
       SUM(Amount) AS Total_Payment
FROM Payment
GROUP BY Payment_Method
HAVING SUM(Amount) > (
    SELECT AVG(Total_Amount)
    FROM (
        SELECT SUM(Amount) AS Total_Amount
        FROM Payment
        GROUP BY Payment_Method
    )
);

--Find customers who have more than one order
SELECT c.Customer_ID,
       c.Customer_Name,
       COUNT(o.Order_ID) AS Total_Orders
FROM Customer c
JOIN Orders o
ON c.Customer_ID = o.Customer_ID
GROUP BY c.Customer_ID, c.Customer_Name
HAVING COUNT(o.Order_ID) > 1;

--Find customers who have placed more orders than the average number of orders per customer
SELECT c.Customer_ID,
       c.Customer_Name,
       COUNT(o.Order_ID) AS Total_Orders
FROM Customer c
JOIN Orders o
ON c.Customer_ID = o.Customer_ID
GROUP BY c.Customer_ID, c.Customer_Name
HAVING COUNT(o.Order_ID) > (
    SELECT AVG(Order_Count)
    FROM (
        SELECT COUNT(*) AS Order_Count
        FROM Orders
        GROUP BY Customer_ID
    )
);

--Find products whose total sold quantity is greater than 5
SELECT p.Product_Name,
       SUM(oi.Quantity) AS Total_Quantity
FROM Product p
JOIN Order_Item oi
ON p.Product_ID = oi.Product_ID
GROUP BY p.Product_Name
HAVING SUM(oi.Quantity) > 5;

--Find customers whose total spending is greater than the average customer spending
SELECT c.Customer_Name,
       SUM(o.Total_Amount) AS Total_Spent
FROM Customer c
JOIN Orders o
ON c.Customer_ID = o.Customer_ID
GROUP BY c.Customer_Name
HAVING SUM(o.Total_Amount) > (
    SELECT AVG(Customer_Total)
    FROM (
        SELECT SUM(Total_Amount) AS Customer_Total
        FROM Orders
        GROUP BY Customer_ID
    )
);
